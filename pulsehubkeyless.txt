-- Hiệu ứng lắc siêu mượt bằng TweenService (Đã tích hợp lại)
local function shakeFrame()
    local originalPos = UDim2.new(0.5, -210, 0.5, -135)
    local tweenInfo = TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    
    local p1 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 8, 0, 0)})
    local p2 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -8, 0, 0)})
    local p3 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 5, 0, 0)})
    local p4 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -5, 0, 0)})
    local p5 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 2, 0, 0)})
    local p6 = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos})

    p1:Play() p1.Completed:Wait()
    p2:Play() p2.Completed:Wait()
    p3:Play() p3.Completed:Wait()
    p4:Play() p4.Completed:Wait()
    p5:Play() p5.Completed:Wait()
    p6:Play()
end
